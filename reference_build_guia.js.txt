const FONT = "Arial";

function p(text, opts = {}) {
  return new Paragraph({
    alignment: opts.align || AlignmentType.JUSTIFIED,
    spacing: { after: opts.after ?? 160, before: opts.before ?? 0 },
    children: [new TextRun({ text, font: FONT, size: (opts.size || 12) * 2, bold: !!opts.bold, italics: !!opts.italics })],
  });
}

function videoApoyoParagraph(v) {
  // Línea "Video de apoyo:" con hipervínculo real a YouTube, justo después de la Introducción.
  return new Paragraph({
    alignment: AlignmentType.JUSTIFIED,
    spacing: { after: 160 },
    children: [
      new TextRun({ text: "Video de apoyo (", font: FONT, size: 24, bold: true }),
      new TextRun({ text: `máx. 5 min — ${v.duracion}): `, font: FONT, size: 24, bold: true }),
      new TextRun({ text: `"${v.titulo}" — ${v.canal}. `, font: FONT, size: 24 }),
      new ExternalHyperlink({
        link: v.url,
        children: [new TextRun({ text: v.url, font: FONT, size: 24, style: "Hyperlink" })],
      }),
    ],
  });
}

function pRuns(runs, opts = {}) {
  return new Paragraph({
    alignment: opts.align || AlignmentType.JUSTIFIED,
    spacing: { after: opts.after ?? 160, before: opts.before ?? 0 },
    indent: opts.indent,
    children: runs.map(r => new TextRun({ text: r.text, font: FONT, size: (opts.size || 12) * 2, bold: !!r.bold, italics: !!r.italics })),
  });
}

function heading(text, opts = {}) {
  return new Paragraph({
    alignment: opts.align || AlignmentType.LEFT,
    spacing: { before: opts.before ?? 300, after: opts.after ?? 160 },
    children: [new TextRun({ text, font: FONT, size: (opts.size || 13) * 2, bold: true, italics: !!opts.italics })],
  });
}

function numItem(n, text) {
  return new Paragraph({
    spacing: { after: 140 },
    indent: { left: 400, hanging: 300 },
    children: [
      new TextRun({ text: `${n}. `, font: FONT, size: 24 }),
      new TextRun({ text, font: FONT, size: 24 }),
    ],
  });
}

function mono(text) {
  // OBSOLETO para "Imagen Representativa" de subtemas — usa rutaVisual() en su lugar (ver
  // scripts/gen_ruta_visual.py y la sección "Imagen Representativa de cada subtema" en SKILL.md).
  // Se deja disponible solo por si se necesita un fragmento de texto monoespaciado puntual.
  return new Paragraph({
    alignment: AlignmentType.LEFT,
    indent: { left: 400 },
    spacing: { after: 40 },
    children: [new TextRun({ text, font: "Courier New", size: 20 })],
  });
}

function rutaVisual(pngPath, width, height) {
  // RESPALDO cuando no hay acceso real a Office (ver capturaReal() abajo, que es el método
  // preferido). Imagen "ruta visual" (pestaña > grupo > íconos) generada con
  // scripts/gen_ruta_visual.py — reemplaza el texto Courier New para "Imagen Representativa:".
  const buf = fs.readFileSync(pngPath);
  return new Paragraph({
    spacing: { after: 160 },
    indent: { left: 400 },
    children: [new ImageRun({ data: buf, type: "png", transformation: { width, height } })],
  });
}

function capturaReal(jpgPath, width = 460, height = 287, producto = "Excel") {
  // MÉTODO PREFERIDO para "Imagen Representativa:" — captura real de Office obtenida vía
  // computer-use (ver sección "Imagen Representativa de cada subtema" en SKILL.md). La imagen debe
  // ser la ventana COMPLETA, solo redimensionada (nunca recortada), por la política de Microsoft.
  // Grande y CENTRADA (no en el margen izquierdo) para que se lea bien. Ancho por defecto ~460px
  // (proporción 460x287, la de una captura de pantalla completa 1389x868). Devuelve [imagen, párrafo
  // de crédito] — hacer children.push(...capturaReal(...)).
  const buf = fs.readFileSync(jpgPath);
  return [
    new Paragraph({
      alignment: AlignmentType.CENTER,
      spacing: { after: 40 },
      children: [new ImageRun({ data: buf, type: "jpg", transformation: { width, height } })],
    }),
    new Paragraph({
      alignment: AlignmentType.CENTER,
      spacing: { after: 160 },
      children: [new TextRun({ text: `Captura de pantalla de Microsoft ${producto}. Used with permission from Microsoft.`, font: FONT, size: 15, italics: true })],
    }),
  ];
}

function iconStep(n, iconFile, text, opts = {}) {
  // opts.native = true (ESTÁNDAR, úsalo siempre que exista el ícono) -> usa el ícono real de
  // Microsoft (Fluent, color gris/negro original #212121, SIN recolorear, SIN insignia de color)
  // desde CONFIG.iconosNativosDir. Ya existe un ícono nativo real para casi cualquier acción típica
  // de Word/Excel/PowerPoint — revisa primero la carpeta 00_PLANTILLAS_REFERENCIA/
  // iconos_pasos_nativos_ms/ antes de asumir que hace falta uno institucional. Solo usa el badge
  // institucional propio (CONFIG.iconosDir, opts.native=false/omitido) como último recurso, cuando
  // de verdad no exista un ícono real de Microsoft para esa acción puntual.
  //
  // IMPORTANTE — coherencia ícono↔texto: cuando busques el ícono más parecido al de Microsoft,
  // ten cuidado con nombres de familia engañosos. Ejemplo real: para "centrar texto/celda" el
  // ícono correcto es "Text Align Center" (3 barras centradas, como el botón real de Word/Excel),
  // NO "Align Center Horizontal" (que en Fluent es para alinear OBJETOS/formas entre sí en
  // PowerPoint, no texto — visualmente es una pesa/dumbbell y confunde). Verifica siempre que el
  // ícono elegido sea el que de verdad aparece en la cinta de opciones para esa acción específica,
  // no solo un ícono con un nombre parecido.
  const dir = opts.native === false ? CONFIG.iconosDir : CONFIG.iconosNativosDir;
  const size = opts.native === false ? 22 : 28;
  const buf = fs.readFileSync(`${dir}/${iconFile}.png`);
  return new Paragraph({
    spacing: { after: 140 },
    indent: { left: 400, hanging: 300 },
    children: [
      new TextRun({ text: `${n}. `, font: FONT, size: 24 }),
      new ImageRun({ data: buf, type: "png", transformation: { width: size, height: size } }),
      new TextRun({ text: `  ${text}`, font: FONT, size: 24 }),
    ],
  });
}

function rcell(text, opts = {}) {
  return new TableCell({
    width: { size: opts.w || 2000, type: WidthType.DXA },
    shading: opts.shade ? { type: ShadingType.CLEAR, fill: opts.shade } : undefined,
    verticalAlign: VerticalAlign.CENTER,
    margins: { top: 80, bottom: 80, left: 80, right: 80 },
    children: [new Paragraph({ alignment: opts.center ? AlignmentType.CENTER : AlignmentType.LEFT, children: [new TextRun({ text, font: FONT, size: 18, bold: !!opts.bold })] })],
  });
}
function buildRubricTable(rows) {
  // rows: array de [criterio, superior, alto, basico, bajo] — la primera fila es el header

    width: { size: 10500, type: WidthType.DXA },
    columnWidths: [1900, 2150, 2150, 2150, 2150],
    rows: rows.map((row, i) => new TableRow({
      children: row.map((text, j) => rcell(text, { w: j === 0 ? 1900 : 2150, bold: i === 0 || j === 0, center: i === 0, shade: i === 0 ? "D9E1F2" : undefined })),
    })),
  });
}
